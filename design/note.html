2. 오프라인 동기화 큐 인터페이스 (SyncQueue Engine)

네트워크 단절 시 모든 변경 트랜잭션은 단일 암호화 로그 스트림으로 버퍼링되며, 재연결 즉시 멱등성(Idempotent) 토큰과 함께 배치 전송됩니다.

SyncCoordinator.ts
TypeScript 
interface OfflineCommitPayload {
  documentId: string;
  vectorClock: Map<string, number>;
  patchOperations: CRDTOperation[];
  idempotencyToken: string;
}

export async function dispatchSyncQueue(
  payload: OfflineCommitPayload
): Promise<SyncAck> {
  /* 1. 로컬 SQLite Commit Lock 획득 */
  const lock = await LocalDB.acquireAtomicLock(payload.documentId);
  try {
    return await CloudChannel.broadcastDelta(payload);
  } finally {
    lock.release();
  }
}
